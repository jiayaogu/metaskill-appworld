def b64url_decode(s):
    alphabet='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_'
    lookup={c:i for i,c in enumerate(alphabet)}
    bits=''
    for c in s:
        if c!='=':
            bits += bin(lookup[c])[2:].zfill(6)
    out=''
    for i in range(0, len(bits)-7, 8):
        out += chr(int(bits[i:i+8],2))
    return out
